(IEnumerable<(string, double)> salesFilesTotals, double) CalculateSalesTotal(IEnumerable<string> salesFiles)
{
    double salesTotal = 0;

    List<(string, double)> filesTotals = new List<(string, double)>();

    // Loop over each file path in salesFiles
    foreach (var file in salesFiles)
    {
        // Read the contents of the file
        string salesJson = File.ReadAllText(file);

        // Parse the contents as JSON
        SalesData? data = JsonConvert.DeserializeObject<SalesData?>(salesJson);

        // Add the file path and the amount found in the Total field to the salesFilesTotals list
        filesTotals.Add((file, data?.Total ?? 0));

        // Add the amount found in the Total field to the salesTotal variable
        salesTotal += data?.Total ?? 0;
    }

    salesFilesTotals = filesTotals;

    return (salesFilesTotals, salesTotal);
}
